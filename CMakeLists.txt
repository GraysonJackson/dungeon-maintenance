cmake_minimum_required(VERSION 3.16)
project(DungeonMaintenance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(TINYXML2 REQUIRED IMPORTED_TARGET tinyxml2)
file(GLOB SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(dungeon-maintenance ${SOURCES})
target_link_libraries(dungeon-maintenance PRIVATE PkgConfig::SDL2 PkgConfig::TINYXML2)
if(MSVC)
    target_compile_options(dungeon-maintenance PRIVATE /W4)
else()
    target_compile_options(dungeon-maintenance PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_command(TARGET dungeon-maintenance POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/data" "$<TARGET_FILE_DIR:dungeon-maintenance>/data")
enable_testing()
add_test(NAME component-demo COMMAND dungeon-maintenance --self-test --data "${CMAKE_CURRENT_SOURCE_DIR}/data")
